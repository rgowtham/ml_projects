#+begin_src python
import kaggle
print(f"Kaggle version: {kaggle.__version__}")

# KAGGLE API AUTHENTICATE
kaggle.api.authenticate()

# Now download the datasets
kaggle.api.competition_download_files(
    competition='playground-series-s6e8', # this is from the URL  - https://www.kaggle.com/competitions/playground-series-s6e8/
    path='./data',
    quiet=False
)
